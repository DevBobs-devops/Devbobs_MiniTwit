\def\pumlfile#1{%
  \directlua{%
    local filename = "\luaescapestring{#1}"
    local f = assert(io.open("diagrams/" .. filename, "rb"))
    local content = f:read("*all")
    f:close()
    local f2 = assert(io.open("../out/" .. filename .. ".ltx", "w"))
    f2:write("\string\\begin{plantuml}\string\n" .. content .. "\string\\end{plantuml}")
    f2:close()
  }%
  \input{"../out/#1.ltx"}}
