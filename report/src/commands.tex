\def\pumlfile#1{%
  %TC:ignore
  \directlua{%
    local filename = "\luaescapestring{#1}"
    local f = assert(io.open("diagrams/" .. filename, "rb"))
    local content = f:read("*all")
    f:close()
    local f2 = assert(io.open(filename .. ".ltx", "w"))
    f2:write("\string\\begin{plantuml}\string\n" .. content .. "\string\\end{plantuml}")
    f2:close()
  }%
  \begin{adjustwidth}{-2.5cm}{-2.5cm}%
  \input{"#1.ltx"}%
  \end{adjustwidth}%
  %TC:endignore 
  }

\newcommand{\code}[1]{%
\texttt{#1}%
}
