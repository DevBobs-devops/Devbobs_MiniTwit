\def\pumlfile#1{%
  %TC:ignore
  \directlua{%
    local filename = "\luaescapestring{#1}"
    local inpath  = "diagrams/" .. filename
    local outpath = filename .. ".ltx"

    local fin = assert(io.open(inpath, "rb"))
    local content = fin:read("*all")
    fin:close()

    local fout = assert(io.open(outpath, "w"))
    fout:write("\string\\begin{plantuml}\string\n" .. content .. "\string\\end{plantuml}")
    fout:close()
  }%
  \input{"\luaescapestring{#1}.ltx"}%
  %TC:endignore
}

\newcommand{\code}[1]{%
\texttt{#1}%
}
